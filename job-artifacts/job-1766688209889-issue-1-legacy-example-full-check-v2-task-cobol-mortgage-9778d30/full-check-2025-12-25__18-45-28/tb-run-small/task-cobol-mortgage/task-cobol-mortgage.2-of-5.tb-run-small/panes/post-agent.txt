> IDENTIFICATION DIVISION.
> PROGRAM-ID. MORTGAGE-PROCESS.
> AUTHOR. CHATGPT.
>
> ENVIRONMENT DIVISION.
> INPUT-OUTPUT SECTION.
> FILE-CONTROL.
>     SELECT INFILE ASSIGN TO "/app/src/INPUT.DAT"
>         ORGANIZATION IS LINE SEQUENTIAL.
>     SELECT OUTFILE ASSIGN TO "/app/data/OUTPUT.DAT"
>         ORGANIZATION IS LINE SEQUENTIAL.
>
> DATA DIVISION.
> FILE SECTION.
> FD INFILE.
> 01 IN-REC PIC X(200).
> FD OUTFILE.
> 01 OUT-REC PIC X(200).
>
> WORKING-STORAGE SECTION.
> 01 WS-ID               PIC X(10).
> 01 WS-NAME             PIC X(30).
> 01 WS-LOAN-AMT-STR     PIC X(10).
> 01 WS-PROP-VALUE-STR   PIC X(10).
> 01 WS-DOWN-PAY-STR     PIC X(10).
> 01 WS-TERM-STR         PIC X(2).
> 01 WS-RATE-STR         PIC X(6).
>
> 01 LOAN-AMOUNT         PIC 9(10)V99 COMP-3.
> 01 PROP-VALUE          PIC 9(10)V99 COMP-3.
> 01 DOWN-PAYMENT        PIC 9(10)V99 COMP-3.
> 01 TERM-MONTHS         PIC 9(2) COMP-3.
> 01 ANNUAL-RATE         PIC 9(3)V99 COMP-3.
>
> 01 MIN-DOWN            PIC 9(10)V99 COMP-3.
> 01 MONTHLY-RATE        PIC 9(5)V9999 COMP-3.
> 01 MONTHLY-PAYMENT    PIC 9(10)V99 COMP-3.
> 01 INTEREST-PART      PIC 9(10)V99 COMP-3.
> 01 PRINCIPAL-PART     PIC 9(10)V99 COMP-3.
> 01 REMAINING-BAL       PIC 9(10)V99 COMP-3.
> 01 MON-COUNTER         PIC 9(3) COMP-3.
>
> 01 REASON-REJ          PIC X(50) VALUE
>    "INSUFFICIENT FUNDS FOR DOWN PAYMENT".
>
> PROCEDURE DIVISION.
>     OPEN INPUT INFILE
>     OPEN OUTPUT OUTFILE.
>     PERFORM UNTIL EOF-FLAG = 1
>         READ INFILE
>             AT END SET EOF-FLAG TO 1
>             NOT AT END
>                 PERFORM PROCESS-RECORD
>         END-READ
>     END-PERFORM.
>     CLOSE INFILE
>     CLOSE OUTFILE.
>     STOP RUN.
>
> PROCESS-RECORD.
>     UNSTRING IN-REC DELIMITED BY ';'
>         INTO WS-ID, WS-NAME, WS-LOAN-AMT-STR,
>              WS-PROP-VALUE-STR, WS-DOWN-PAY-STR,
>              WS-TERM-STR, WS-RATE-STR.
>     MOVE FUNCTION NUMVAL(WS-LOAN-AMT-STR) TO LOAN-AMOUNT.
>     MOVE FUNCTION NUMVAL(WS-PROP-VALUE-STR) TO PROP-VALUE.
>     MOVE FUNCTION NUMVAL(WS-DOWN-PAY-STR) TO DOWN-PAYMENT.
>     MOVE FUNCTION NUMVAL(WS-TERM-STR) TO TERM-MONTHS.
>     MOVE FUNCTION NUMVAL(WS-RATE-STR) TO ANNUAL-RATE.
>
>     COMPUTE MIN-DOWN = PROP-VALUE * 0.15 ROUNDED.
>     IF DOWN-PAYMENT < MIN-DOWN THEN
>         STRING WS-ID DELIMITED BY SIZE
>                " "
>                REASON-REJ DELIMITED BY SIZE
>                INTO OUT-REC
>         END-STRING
>         WRITE OUT-REC
>     ELSE
>         STRING WS-ID DELIMITED BY SIZE
>                " APPRO"
>                INTO OUT-REC
>         END-STRING
>         WRITE OUT-REC
>         PERFORM CALCULATE-SCHEDULE
>     END-IF.
>
> CALCULATE-SCHEDULE.
>     COMPUTE MONTHLY-RATE = ANNUAL-RATE / 1200 ROUNDED.
>     COMPUTE MONTHLY-PAYMENT = LOAN-AMOUNT * MONTHLY-RATE /
>         (1 - FUNCTION POWER(1 + MONTHLY-RATE, -TERM-MONTHS))
>         ROUNDED.
>     MOVE LOAN-AMOUNT TO REMAINING-BAL.
>     PERFORM VARYING MON-COUNTER FROM 1 BY 1 UNTIL MON-COUNTER > TERM-MONTHS
>         COMPUTE INTEREST-PART = REMAINING-BAL * MONTHLY-RATE ROUNDED.
>         COMPUTE PRINCIPAL-PART = MONTHLY-PAYMENT - INTEREST-PART ROUNDED.
>         SUBTRACT PRINCIPAL-PART FROM REMAINING-BAL.
>         STRING MON-COUNTER DELIMITED BY SIZE
>                " "
>                FUNCTION TRIM(FUNCTION DISPLAY(MONTHLY-PAYMENT, "ZZZZZZZ9.99"))
>                " "
>                FUNCTION TRIM(FUNCTION DISPLAY(INTEREST-PART, "ZZZZZZZ9.99"))
>                " "
>                FUNCTION TRIM(FUNCTION DISPLAY(PRINCIPAL-PART, "ZZZZZZZ9.99"))
>                " "
>                FUNCTION TRIM(FUNCTION DISPLAY(REMAINING-BAL, "ZZZZZZZZZZ9.99"))
>                INTO OUT-REC
>         END-STRING
>         WRITE OUT-REC
>     END-PERFORM.
>
> EOF
