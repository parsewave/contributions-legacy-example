>         END-STRING
>         WRITE OUTPUT-REC

>         COMPUTE WS-MONTHLY-PAYMENT = WS-LOAN-AMOUNT * WS-MONTHLY-RATE /
>             (1 - FUNCTION POWER(1 + WS-MONTHLY-RATE, -WS-TERM)) ROUNDED
>         MOVE WS-LOAN-AMOUNT TO WS-BALANCE
>         PERFORM VARYING WS-MONTH-NUM FROM 1 BY 1 UNTIL WS-MONTH-NUM > WS-TERM
>             COMPUTE WS-INTEREST = WS-BALANCE * WS-MONTHLY-RATE ROUNDED
>             COMPUTE WS-PRINCIPAL = WS-MONTHLY-PAYMENT - WS-INTEREST ROUNDED
>             COMPUTE WS-BALANCE = WS-BALANCE - WS-PRINCIPAL ROUNDED
>             * Prepare formatted fields
>             MOVE WS-MONTH-NUM TO WS-MONTH-NUM-ED
>             MOVE WS-MONTHLY-PAYMENT TO WS-MONTHLY-PAYMENT-ED
>             MOVE WS-INTEREST TO WS-INTEREST-ED
>             MOVE WS-PRINCIPAL TO WS-PRINCIPAL-ED
>             MOVE WS-BALANCE TO WS-BALANCE-ED
>             STRING WS-MONTH-NUM-ED " "

>                    WS-INTEREST-ED " "
>                    WS-PRINCIPAL-ED " "
>                    WS-BALANCE-ED DELIMITED BY SIZE
>                    INTO OUTPUT-REC
>             END-STRING
>             WRITE OUTPUT-REC
>         END-PERFORM
>     END-IF.
